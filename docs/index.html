Integrate into Any Solana AI Agent

import { SolanaAgentTreasury } from 'solana-agent-treasury';

// 1. Initialize Treasury with a $30.00 safety reserve
const treasury = new SolanaAgentTreasury({
  reserveFloorUsdc: 30.00,
});

// 2. Automatically vault operating reserve into Kamino Lending
const status = treasury.getReserveStatus();
console.log(`Kamino APY: ${status.currentApy}% | Deposited: $${status.depositedUsdc}`);

// 3. Scan & trade dips on Jupiter with Capital Guard protection
const signal = treasury.evaluateDip('SOL', mintAddress, currentPrice, rsi, change24h, walletBalance);
if (signal.shouldBuy) {
  const position = treasury.executeDipTrade(signal);
  console.log(`Bought ${position.amount} SOL at $${position.entryPrice}`);
}
